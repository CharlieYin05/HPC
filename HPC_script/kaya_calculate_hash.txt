创建通用求解脚本

cd ~/Assignment2_HPC
mkdir -p logs runs output/assignment2

cat > solve_remaining.slurm <<'EOF'
#!/bin/bash
#SBATCH --job-name=a2_solve
#SBATCH --partition=cits3402
#SBATCH --nodes=1
#SBATCH --ntasks-per-node=8
#SBATCH --cpus-per-task=8
#SBATCH --time=00:30:00
#SBATCH --output=logs/%x_%j.out
#SBATCH --error=logs/%x_%j.err

set -euo pipefail

module load gcc/14.3.0
module load openmpi/5.0.5

ROOT="$SLURM_SUBMIT_DIR"

: "${PAIR:?PAIR must be set}"
: "${NUMBER:?NUMBER must be set}"

# 每个作业使用独立目录，避免覆盖其他作业的结果
WORK="$ROOT/runs/${PAIR}_${SLURM_JOB_ID}"
mkdir -p "$WORK/output" "$ROOT/output/assignment2"
cd "$WORK"

export OMP_NUM_THREADS=8
export OMP_PROC_BIND=close
export OMP_PLACES=cores

echo "Pair: $PAIR"
echo "Job ID: $SLURM_JOB_ID"
echo "Nodes: $SLURM_JOB_NUM_NODES"
echo "MPI ranks: $SLURM_NTASKS"
echo "Threads per rank: $OMP_NUM_THREADS"
echo "Start: $(date)"

mpirun --map-by "slot:PE=8" --bind-to core \
    -np "$SLURM_NTASKS" "$ROOT/birthday_hybrid" \
    "$ROOT/data/assignment2/${NUMBER}_${PAIR}_a.pdf" \
    "$ROOT/data/assignment2/${NUMBER}_${PAIR}_b.pdf" \
    23497868 50000000 \
    --threads 8 \
    --batch 16384 \
    --collisions 1

# 确认文件存在后，保存到最终输出目录
test -s output/hybrid_a.pdf
test -s output/hybrid_b.pdf

cp output/hybrid_a.pdf \
   "$ROOT/output/assignment2/${NUMBER}_${PAIR}_a_solved.pdf"

cp output/hybrid_b.pdf \
   "$ROOT/output/assignment2/${NUMBER}_${PAIR}_b_solved.pdf"

echo "SUCCESS: $PAIR solved PDFs saved."
echo "End: $(date)"
EOF

创建一键提交脚本

cat > submit_remaining.sh <<'EOF'
#!/bin/bash
set -euo pipefail

cd ~/Assignment2_HPC
mkdir -p logs runs output/assignment2

# 编号 文件名 最长运行时间
configs=(
    "3 gamma 00:30:00"
    "4 delta 00:45:00"
    "5 epsilon 01:00:00"
    "6 zeta 01:50:00"
)

previous=""

for config in "${configs[@]}"; do
    read -r number pair limit <<< "$config"

    dependency=()
    if [ -n "$previous" ]; then
        dependency=("--dependency=afterany:$previous")
    fi

    job_id=$(sbatch --parsable \
        --job-name="solve_${pair}" \
        --time="$limit" \
        --export="ALL,NUMBER=$number,PAIR=$pair" \
        "${dependency[@]}" \
        solve_remaining.slurm)

    echo "$pair -> Job ID $job_id (limit $limit)"
    previous="$job_id"
done

echo "All four searches submitted."
EOF
